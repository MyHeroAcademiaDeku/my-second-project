import os  
import sys  
import shutil  
import traceback  
import json  
import urllib.request  
import urllib.parse  
import urllib.error  
  
TARGET = "geo_core.py"          # must match your renamed core file  
BACKUP = "geo_core.py.bak"  
MAX_ATTEMPTS = 4  
  
# Uses the SAME key your geo_core.py uses, so you only set one variable.  
OPENROUTER_URL = "https://openrouter.ai/api/v1/chat/completions"  
OPENROUTER_MODEL = "meta-llama/llama-3.3-70b-instruct:free"  # a :free model  
  
  
def ask_llm_to_fix(broken_source, error_text):  
    """Send the broken core + traceback to OpenRouter and get corrected source back."""  
    key = os.environ.get("OPENROUTER_API_KEY", "")  
    if not key:  
        print("   [Repair debug] No OPENROUTER_API_KEY set — cannot AI-repair.")  
        return None  
  
    prompt = (  
        "You are a Python repair tool. The following module crashed. "  
        "Return ONLY the complete corrected Python source with no markdown "  
        "fences, no explanation, and no commentary.\n\n"  
        f"--- TRACEBACK ---\n{error_text}\n\n"  
        f"--- SOURCE ---\n{broken_source}"  
    )  
  
    body = json.dumps({  
        "model": OPENROUTER_MODEL,  
        "messages": [{"role": "user", "content": prompt}],  
    }).encode("utf-8")  
  
    req = urllib.request.Request(OPENROUTER_URL, data=body, method="POST")  
    req.add_header("Authorization", f"Bearer {key}")  
    req.add_header("Content-Type", "application/json")  
  
    try:  
        with urllib.request.urlopen(req, timeout=60) as response:  
            raw = response.read().decode("utf-8")  
        data = json.loads(raw)  
        fixed = data["choices"][0]["message"]["content"].strip()  
        # Strip accidental code fences if the model added them anyway.  
        if fixed.startswith("```"):  
            fixed = fixed.split("```", 2)[1]  
            if fixed.startswith("python"):  
                fixed = fixed[len("python"):]  
            fixed = fixed.strip("`").strip()  
        return fixed if fixed else None  
    except urllib.error.HTTPError as e:  
        try:  
            print(f"   [Repair debug] HTTP {e.code}: {e.read().decode('utf-8')[:300]}")  
        except Exception:  
            print(f"   [Repair debug] HTTP {e.code}: {e}")  
        return None  
    except Exception as e:  
        print(f"   [Repair debug] {type(e).__name__}: {e}")  
        return None  
  
  
def master_fail_safe_boot():  
    print("============================================================")  
    print("🛡️  AI SELF-REPAIR SHIELD: INITIALIZING BOOT LOAD")  
    print("============================================================")  
  
    if not os.path.exists(TARGET):  
        print(f"❌ Error: '{TARGET}' was not found. Put it in this folder.")  
        return  
  
    # Always back up before we ever overwrite the core.  
    shutil.copyfile(TARGET, BACKUP)  
    print(f"💾 Backup saved to '{BACKUP}'.")  
  
    for attempt in range(1, MAX_ATTEMPTS + 1):  
        print(f"\n🕵️  Attempt {attempt}/{MAX_ATTEMPTS}: validating and launching core...")  
        try:  
            with open(TARGET, "r", encoding="utf-8") as f:  
                code_content = f.read()  
  
            compiled = compile(code_content, TARGET, "exec")  
            runtime_scope = {"__name__": "__main__"}  
            exec(compiled, runtime_scope)  
            return  # clean exit — core ran and returned normally  
  
        except Exception:  
            error_text = traceback.format_exc()  
            print("\n💥 [CRASH INTERCEPTED]:")  
            print(error_text)  
  
            if attempt == MAX_ATTEMPTS:  
                print("\n🔁 Attempts exhausted — restoring the original core from backup.")  
                shutil.copyfile(BACKUP, TARGET)  
                print(f"↩️  Restored '{TARGET}' from '{BACKUP}'.")  
                return  
  
            print("\n🤖 Requesting an AI fix from OpenRouter...")  
            fixed_source = ask_llm_to_fix(code_content, error_text)  
            if not fixed_source:  
                print("⚠️  No usable fix returned — restoring backup and stopping.")  
                shutil.copyfile(BACKUP, TARGET)  
                return  
  
            with open(TARGET, "w", encoding="utf-8") as f:  
                f.write(fixed_source)  
            print("✅ Wrote AI-corrected core. Retrying...")  
  
  
if __name__ == "__main__":  
    master_fail_safe_boot()
